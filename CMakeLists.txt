cmake_minimum_required(VERSION 3.21.1)
project(HelixEdit VERSION 0.1.4 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Widgets Svg Test)
add_library(helix_core src/sequence.cpp)
target_include_directories(helix_core PUBLIC src)
target_link_libraries(helix_core PUBLIC Qt6::Core)
qt_add_executable(helixedit MACOSX_BUNDLE src/main.cpp src/editor.cpp src/editor.h src/wrapped_sequence.cpp src/wrapped_sequence.h)
qt_add_resources(helixedit app_assets PREFIX "/" FILES packaging/helixedit.svg)
set(HELIX_SAMPLE_FASTA
  examples/samples/dna_orchid_6.fasta
  examples/samples/rna_transcripts_3.fasta
  examples/samples/proteins_5.fasta
)
qt_add_resources(helixedit example_sequences PREFIX "/" FILES ${HELIX_SAMPLE_FASTA})
target_link_libraries(helixedit PRIVATE helix_core Qt6::Gui Qt6::Widgets Qt6::Svg)
set_target_properties(helixedit PROPERTIES MACOSX_BUNDLE_GUI_IDENTIFIER org.helixedit.desktop MACOSX_BUNDLE_BUNDLE_NAME HelixEdit MACOSX_BUNDLE_SHORT_VERSION_STRING ${PROJECT_VERSION})
include(CTest)
if(BUILD_TESTING)
  qt_add_executable(core_tests tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE helix_core Qt6::Test)
  add_test(NAME sequence_core COMMAND core_tests)
  qt_add_executable(orf_tests tests/orf_tests.cpp)
  target_link_libraries(orf_tests PRIVATE helix_core Qt6::Test)
  add_test(NAME sequence_orf COMMAND orf_tests)
  qt_add_executable(restriction_tests tests/restriction_tests.cpp)
  target_link_libraries(restriction_tests PRIVATE helix_core Qt6::Test)
  add_test(NAME plasmid_restriction COMMAND restriction_tests)
  qt_add_executable(primer_tests tests/primer_tests.cpp)
  target_link_libraries(primer_tests PRIVATE helix_core Qt6::Test)
  add_test(NAME plasmid_primers COMMAND primer_tests)
  qt_add_executable(academic_tests tests/academic_tests.cpp)
  target_link_libraries(academic_tests PRIVATE helix_core Qt6::Test)
  add_test(NAME academic_workflows COMMAND academic_tests)
  qt_add_executable(statistics_tests tests/statistics_tests.cpp)
  target_link_libraries(statistics_tests PRIVATE helix_core Qt6::Test)
  add_test(NAME analysis_statistics COMMAND statistics_tests)
  qt_add_executable(editor_tests tests/editor_tests.cpp src/editor.cpp src/editor.h src/wrapped_sequence.cpp src/wrapped_sequence.h)
  qt_add_resources(editor_tests example_test_sequences PREFIX "/" FILES ${HELIX_SAMPLE_FASTA})
  qt_add_resources(editor_tests icon_test_assets PREFIX "/" FILES packaging/helixedit.svg)
  target_link_libraries(editor_tests PRIVATE helix_core Qt6::Test Qt6::Gui Qt6::Widgets Qt6::Svg)
  add_test(NAME editor_workflows COMMAND editor_tests)
  set_tests_properties(editor_workflows PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 120)
endif()
include(GNUInstallDirs)
install(FILES LICENSE DESTINATION share/doc/helixedit)
install(DIRECTORY examples/samples/ DESTINATION share/helixedit/examples/samples FILES_MATCHING PATTERN "*.fasta" PATTERN "README.md")
install(TARGETS helixedit BUNDLE DESTINATION . RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
if(UNIX AND NOT APPLE)
  install(FILES packaging/org.helixedit.desktop DESTINATION share/applications)
  install(PROGRAMS scripts/update_from_source.sh DESTINATION share/helixedit/scripts)
  install(FILES packaging/helixedit.svg DESTINATION share/icons/hicolor/scalable/apps)
  install(FILES packaging/helixedit.svg DESTINATION share/pixmaps)
  configure_file(packaging/postinst.in "${CMAKE_CURRENT_BINARY_DIR}/postinst" COPYONLY)
  file(CHMOD "${CMAKE_CURRENT_BINARY_DIR}/postinst"
    PERMISSIONS OWNER_READ OWNER_WRITE OWNER_EXECUTE GROUP_READ GROUP_EXECUTE WORLD_READ WORLD_EXECUTE)
  set(CPACK_DEBIAN_PACKAGE_CONTROL_EXTRA "${CMAKE_CURRENT_BINARY_DIR}/postinst")
  set(CPACK_GENERATOR DEB)
  set(CPACK_PACKAGE_CONTACT "HelixEdit contributors")
  set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Compact sequence and alignment editor")
  set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
  set(CPACK_DEBIAN_PACKAGE_NAME helixedit)
  set(CPACK_PACKAGE_FILE_NAME "HelixEdit-${PROJECT_VERSION}-${CMAKE_SYSTEM_PROCESSOR}")
  set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
  include(CPack)
endif()
